<script>
  (function () {
    var storageKey = "ke-homepage-language";
    var currentLang = "en";
    var toggle = document.querySelector("[data-lang-toggle]");
    var toggleText = document.querySelector("[data-lang-toggle-text]");

    function getStoredLang() {
      try {
        return localStorage.getItem(storageKey);
      } catch (error) {
        return null;
      }
    }

    function storeLang(lang) {
      try {
        localStorage.setItem(storageKey, lang);
      } catch (error) {
        return;
      }
    }

    function applyLang(lang) {
      currentLang = lang === "zh" ? "zh" : "en";
      document.documentElement.lang = currentLang === "zh" ? "zh-CN" : "en-US";
      document.body.classList.toggle("is-zh", currentLang === "zh");

      document.querySelectorAll("[data-i18n-en][data-i18n-zh]").forEach(function (element) {
        var nextText = element.getAttribute("data-i18n-" + currentLang);
        if (nextText !== null) {
          element.innerHTML = nextText;
        }
      });

      if (toggleText) {
        toggleText.textContent = currentLang === "zh" ? "English" : "中文";
      }

      if (toggle) {
        toggle.setAttribute("aria-label", currentLang === "zh" ? "Switch to English" : "切换到中文");
      }

      storeLang(currentLang);
    }

    applyLang(getStoredLang() || "en");

    if (toggle) {
      toggle.addEventListener("click", function () {
        applyLang(currentLang === "zh" ? "en" : "zh");
      });
    }
  })();
</script>
